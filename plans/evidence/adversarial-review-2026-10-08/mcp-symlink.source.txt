import { createMcpServer } from '@yumcss/analyze/mcp';
import { createRequire } from 'node:module';
import { PassThrough } from 'node:stream';
import { mkdtempSync, mkdirSync, writeFileSync, readFileSync, symlinkSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
const require=createRequire('/Users/astahmer/dev/pandwind/packages/analyze/package.json');
const {StdioServerTransport}=require('@modelcontextprotocol/server/stdio');
const fixture=mkdtempSync(join(tmpdir(),'yumcss-symlink-'));const root=join(fixture,'project');mkdirSync(root);
const outside=join(fixture,'outside.tsx');
const source='import {css} from "@emotion/css"; export const cls=css({color:"red"});';
writeFileSync(outside,source);symlinkSync(outside,join(root,'Card.tsx'));
const input=new PassThrough();const output=new PassThrough();const server=createMcpServer(root);
let buffer='';const requests=new Map();output.on('data',chunk=>{buffer+=chunk;let index;while((index=buffer.indexOf('\n'))!==-1){const value=JSON.parse(buffer.slice(0,index));buffer=buffer.slice(index+1);requests.get(value.id)?.(value);}});
const call=(id,method,params)=>new Promise(resolve=>{requests.set(id,resolve);input.write(JSON.stringify({jsonrpc:'2.0',id,method,params})+'\n');});
await server.connect(new StdioServerTransport(input,output));
await call(1,'initialize',{protocolVersion:'2025-11-25',capabilities:{},clientInfo:{name:'audit',version:'1'}});
const response=await call(2,'tools/call',{name:'apply_migration',arguments:{files:['Card.tsx'],from:'emotion',to:'tailwind',confirm:true}});
console.log(JSON.stringify({isError:response.result?.isError,error:response.error,externalBefore:source,externalAfter:readFileSync(outside,'utf8'),result:response.result?.structuredContent},null,2));
await server.close();
